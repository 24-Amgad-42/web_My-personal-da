#!/usr/bin/env python3
import socket
import sys
from datetime import datetime

try:
    from scapy.all import ARP, Ether, srp
except ImportError:
    print("[-] Scapy library is not installed.")
    print("[*] Please install it using: pip install scapy")
    sys.exit(1)

COMMON_PORTS = {
    21: {"service": "FTP", "vulnerability": "Legacy systems may allow Anonymous Login or Remote Code Execution."},
    22: {"service": "SSH", "vulnerability": "Vulnerable to Brute Force attacks if weak passwords are used."},
    23: {"service": "Telnet", "vulnerability": "Unencrypted protocol; credentials and data are transmitted in cleartext."},
    80: {"service": "HTTP", "vulnerability": "May host web applications vulnerable to XSS, SQLi, or misconfigurations."},
    443: {"service": "HTTPS", "vulnerability": "Check for outdated SSL/TLS versions (e.g., SSLv3) or expired certificates."},
    445: {"service": "SMB", "vulnerability": "Critical risk; susceptible to exploits like EternalBlue (MS17-010) if unpatched."}
}

def scan_local_network(ip_range):
    print(f"\n[*] Starting network scan for range: {ip_range}")
    try:
        arp = ARP(pdst=ip_range)
        ether = Ether(dst="ff:ff:ff:ff:ff:ff")
        packet = ether/arp

        result = srp(packet, timeout=3, verbose=0)[0]

        devices = []
        for sent, received in result:
            devices.append({'ip': received.psrc, 'mac': received.hwsrc})
        return devices
    except PermissionError:
        print("[-] Permission Error: Script must be run with Administrator / Root privileges.")
        sys.exit(1)
    except Exception as e:
        print(f"[-] An unexpected error occurred during host discovery: {e}")
        return []

def scan_ports(ip):
    print(f"\n[+] Starting port scan for host: {ip}")
    open_ports_found = False
    
    for port, info in COMMON_PORTS.items():
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        s.settimeout(1.0)
        
        result = s.connect_ex((ip, port))
        if result == 0:
            open_ports_found = True
            print(f"   [!] Open Port {port} -> Service: ({info['service']})")
            print(f"       ⚠️ Potential Risk: {info['vulnerability']}")
        s.close()
        
    if not open_ports_found:
        print("   [-] No common open ports found on this host.")

def main():
    print("=" * 65)
    print("        Network Scanner & Vulnerability Assesser (v1.0)        ")
    print("=" * 65)
    
    target_network = input("Enter the network range to scan (e.g., 192.168.1.1/24): ").strip()
    
    if not target_network:
        print("[-] Network range cannot be empty.")
        return

    start_time = datetime.now()
    
    devices_list = scan_local_network(target_network)
    
    print(f"\n[+] Detected ({len(devices_list)}) active hosts on the network:")
    for device in devices_list:
        print(f"   - IP Address: {device['ip']} \t MAC Address: {device['mac']}")
    
    print("-" * 65)
    
    if devices_list:
        run_port_scan = input("\nDo you want to scan open ports and risks for detected hosts? (y/n): ").strip().lower()
        if run_port_scan == 'y':
            for device in devices_list:
                scan_ports(device['ip'])
    else:
        print("[-] No active hosts found to perform port scanning.")
        
    end_time = datetime.now()
    print(f"\n[*] Scan completed successfully. Duration: {end_time - start_time}")

if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print("\n[-] Scan aborted by the user.")
        sys.exit(0)
